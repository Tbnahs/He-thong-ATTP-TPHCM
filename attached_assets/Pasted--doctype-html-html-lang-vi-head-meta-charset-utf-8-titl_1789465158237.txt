/* ================= DATA GENERATION (mock, deterministic) ================= */
function mulberry32(seed){
  return function(){
    seed |= 0; seed = (seed + 0x6D2B79F5) | 0;
    let t = Math.imul(seed ^ (seed >>> 15), 1 | seed);
    t = (t + Math.imul(t ^ (t >>> 7), 61 | t)) ^ t;
    return ((t ^ (t >>> 14)) >>> 0) / 4294967296;
  };
}
const rnd = mulberry32(20260908);
function pick(arr){ return arr[Math.floor(rnd()*arr.length)]; }
function randInt(a,b){ return Math.floor(a + rnd()*(b-a+1)); }

const WARDS = ["Bến Nghé","Bến Thành","Cầu Kho","Cầu Ông Lãnh","Cô Giang","Đa Kao","Nguyễn Cư Trinh","Nguyễn Thái Bình","Phạm Ngũ Lão","Tân Định"];

  "Cấp 2": ["Trần Văn Ơn","Nguyễn Du","Lê Quý Đôn","Võ Trường Toản","Colette","Trần Hưng Đạo"],

// ---- suppliers: 28 total, capacity sums ~18,500, serving sums ~16,800 ----
const N_SUPPLIERS = 28;
let suppliers = [];
let capAcc = 0, servAcc = 0, schoolLinkAcc = 0;
for(let i=0;i<N_SUPPLIERS;i++){
  const capacity = randInt(280,1300);
  suppliers.push({
    id:"S"+(i+1),
    name:"Công ty TNHH "+pick(SUPPLIER_NAMES)+(rnd()>0.6?" "+pick(["Sài Gòn","Q1","Việt Nam","Foods"]):""),
    ward: pick(WARDS),
    address: randInt(2,220)+" "+pick(STREETS),
    capacity,
    schools: randInt(1,7)
  });
  capAcc += capacity;
}
// scale capacities to sum to 18500
const capScale = 18500/capAcc;
suppliers.forEach(s=>{ s.capacity = Math.round(s.capacity*capScale); });
capAcc = suppliers.reduce((a,s)=>a+s.capacity,0);
// diff fix on last item

/* ================= FILTERING (always full-set based) ================= */
function filteredSuppliers(){
  return suppliers.filter(s=>{
    if(state.phuong!=="all" && s.ward!==state.phuong) return false;
    if(state.status!=="all" && s.status!==state.status) return false;
    if(state.search && !s.name.toLowerCase().includes(state.search)) return false;
    return true;
  });
}
function filteredSchools(){
  return schools.filter(e=>{
    if(state.phuong!=="all" && e.ward!==state.phuong) return false;
    if(state.capho!=="all" && e.level!==state.capho) return false;
    if(state.hinhthuc!=="all" && e.mode!==state.hinhthuc) return false;
    if(state.status!=="all" && e.status!==state.status) return false;
    if(state.search && !e.name.toLowerCase().includes(state.search)) return false;
    return true;
  });
}

/* ================= RENDER ================= */
function statusBreakdown(list){
  const pass=list.filter(x=>x.status==="pass").length;
  const warn=list.filter(x=>x.status==="warn").length;
  const fail=list.filter(x=>x.status==="fail").length;
  const total=list.length || 1;
  return {pass,warn,fail,total,rate: list.length? Math.round(pass/list.length*100):0};
}
function renderStatus(list, trackId, legendId){
  const b = statusBreakdown(list);
  const track = document.getElementById(trackId);
  track.innerHTML = "";
  if(b.total>0){
    const segs = [["seg-good",b.pass],["seg-warn",b.warn],["seg-bad",b.fail]];
    segs.forEach(([cls,count])=>{
      if(count>0){
        const el=document.createElement("div");
        el.className=cls;
        el.style.width=(count/list.length*100)+"%";
        track.appendChild(el);
      }
    });
  }
  document.getElementById(legendId).innerHTML = `
    <div class="legend-item"><span class="dot" style="background:var(--good)"></span>Đạt <b>${fmt(b.pass)}</b></div>
    <div class="legend-item"><span class="dot" style="background:var(--warn)"></span>Cảnh báo <b>${fmt(b.warn)}</b></div>
    <div class="legend-item"><span class="dot" style="background:var(--bad)"></span>Chưa đạt <b>${fmt(b.fail)}</b></div>
    <div class="rate-pill">Tỷ lệ đạt &nbsp;<b>${list.length? b.rate+"%":"—"}</b></div>
  `;
}
function badge(status){
  if(status==="pass") return '<span class="badge badge-pass">ĐẠT</span>';
  if(status==="warn") return '<span class="badge badge-warn">CẢNH BÁO</span>';
  return '<span class="badge badge-fail">CHƯA ĐẠT</span>';
}
